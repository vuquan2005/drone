cmake_minimum_required(VERSION 3.22)
project(mixer_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
set(FIRMWARE_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_executable(mixer_tests test_mixer.c ${FIRMWARE_ROOT}/Control/Src/mixer.c)
target_include_directories(mixer_tests PRIVATE ${FIRMWARE_ROOT}/Control/Inc)
target_compile_options(mixer_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(mixer_tests PRIVATE m)
add_test(NAME mixer_tests COMMAND mixer_tests)
