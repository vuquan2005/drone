\chapter*{DANH MỤC CHỮ VIẾT TẮT}
\addcontentsline{toc}{chapter}{DANH MỤC CHỮ VIẾT TẮT}

\begin{tabular}{@{}L{4cm}L{9cm}@{}}
	\textbf{Chữ viết tắt} & \textbf{Ý nghĩa}                                                                                    \\
	UAV                   & Unmanned Aerial Vehicle -- Thiết bị bay không người lái                                             \\
	IMU                   & Inertial Measurement Unit -- Đơn vị đo lường quán tính                                              \\
	ESC                   & Electronic Speed Controller -- Bộ điều khiển tốc độ động cơ                                         \\
	PID                   & Proportional--Integral--Derivative -- Tỉ lệ--tích phân--vi phân                                     \\
	PWM                   & Pulse Width Modulation -- Điều chế độ rộng xung                                                     \\
	RC                    & Radio Control -- Điều khiển vô tuyến                                                                \\
	SPI                   & Serial Peripheral Interface -- Giao tiếp ngoại vi nối tiếp                                          \\
	USART                 & Universal Synchronous/Asynchronous Receiver/Transmitter -- Bộ thu phát nối tiếp đồng bộ/bất đồng bộ \\
	DMA                   & Direct Memory Access -- Truy cập bộ nhớ trực tiếp                                                   \\
	CRSF                  & Crossfire -- Giao thức nối tiếp cho bộ thu điều khiển                                               \\
	CRC                   & Cyclic Redundancy Check -- Kiểm tra dư vòng                                                         \\
	HAL                   & Hardware Abstraction Layer -- Lớp trừu tượng phần cứng                                              \\
	SWD                   & Serial Wire Debug -- Giao tiếp nạp và gỡ lỗi nối tiếp                                               \\
	NED                   & North--East--Down -- Bắc--Đông--Xuống                                                               \\
	FRD                   & Forward--Right--Down -- Trước--Phải--Xuống                                                          \\
	CW                    & Clockwise -- Cùng chiều kim đồng hồ                                                                 \\
	CCW                   & Counterclockwise -- Ngược chiều kim đồng hồ                                                         \\
	RMSE                  & Root Mean Square Error -- Căn quân phương của sai lệch                                              \\
\end{tabular}
